        card.addEventListener('click', function(e) {
          e.preventDefault();
          const featureId = this.getAttribute('data-feature');
          const data = featureData[featureId];
          if (data) {
            document.getElementById('modalTitleSpan').innerHTML = data.title;
            const iconHTML = document.getElementById(data.icon);
            if (iconHTML) {
               document.getElementById('modalIcon').innerHTML = iconHTML.innerHTML;
            }

            modal.classList.add('active');
          }

      // Закрытие модалки
      closeBtn.addEventListener('click', (e) => {

        if (e.target === modal) {
          modal.classList.remove('active');
        }
      });

      // Убираем скроллбары в iframe и добавляем drag-to-scroll
      setTimeout(() => {
        document.querySelectorAll('iframe').forEach(iframe => {
            try {
              const doc = iframe.contentDocument || iframe.contentWindow.document;
              
              // Hide scrollbars
              const style = doc.createElement('style');
              style.textContent = `
                ::-webkit-scrollbar { width: 0px !important; height: 0px !important; background: transparent !important; }
                * { -ms-overflow-style: none; scrollbar-width: none; }
              `;
              doc.head.appendChild(style);

              // Add Drag-to-scroll logic for horizontal scrolling elements
              let isDown = false;
              let startX;
              let scrollLeft;
              let activeContainer = null;

              doc.addEventListener('mousedown', (e) => {
                // Find closest scrollable container
                let target = e.target;
                while (target && target !== doc.body) {
                  const overflowX = iframe.contentWindow.getComputedStyle(target).overflowX;
                  if ((overflowX === 'auto' || overflowX === 'scroll') && target.scrollWidth > target.clientWidth) {
                    activeContainer = target;
                    isDown = true;
                    startX = e.pageX - activeContainer.offsetLeft;
                    scrollLeft = activeContainer.scrollLeft;
                    activeContainer.style.cursor = 'grabbing';
                    activeContainer.style.scrollBehavior = 'auto'; // disable smooth scroll while dragging
                    break;
                  }
                  target = target.parentElement;
                }
              });

              doc.addEventListener('mouseleave', () => {
                if (!isDown) return;
                isDown = false;
                if (activeContainer) activeContainer.style.cursor = 'default';
              });

              doc.addEventListener('mouseup', () => {
                if (!isDown) return;
                isDown = false;
                if (activeContainer) {
                  activeContainer.style.cursor = 'default';
                  activeContainer.style.scrollBehavior = 'smooth';
                }
              });

              doc.addEventListener('mousemove', (e) => {
                if (!isDown || !activeContainer) return;
                e.preventDefault();
                const x = e.pageX - activeContainer.offsetLeft;
                const walk = (x - startX) * 2; // scroll-fast multiplier
                activeContainer.scrollLeft = scrollLeft - walk;
              });

            } catch(e) {
              console.error('Iframe drag-to-scroll error:', e);
            }
        });
      }, 2000);
    });
